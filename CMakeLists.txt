cmake_minimum_required(VERSION 3.22)
project(CamachoLiveSoundProcessor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
juce_add_plugin(CamachoLiveSoundProcessor COMPANY_NAME "Camacho Audio" BUNDLE_ID "uy.camacho.livesound" PLUGIN_MANUFACTURER_CODE Cmch PLUGIN_CODE Clsp FORMATS VST3 PRODUCT_NAME "CAMACHO LIVE SOUND PROCESSOR" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(CamachoLiveSoundProcessor PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(CamachoLiveSoundProcessor PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(CamachoLiveSoundProcessor PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
